# CBF-INC-RRT 常用命令清单
# 说明：
# 1. 所有命令默认在仓库根目录运行：/Users/frankcui/Desktop/CBF-INC-RRT
# 2. 训练依赖里要求 Python 3.9；不要用当前 base 里的 Python 3.13 跑训练。
# 3. 如果机器没有 GPU，把配置里的 device 改成 cpu，或直接让脚本自动走 CPU；速度会慢很多。
# 4. 下面的 PATH/CKPT/HParams 路径需要按你实际训练输出替换。


# =========================
# 0. 环境安装
# =========================

# 创建并激活 conda 环境
conda create --name cbf_inc_rrt python=3.9 -y
conda activate cbf_inc_rrt

# 安装本仓库和依赖
pip install -e .
pip install -r requirements.txt

# 检查关键依赖是否可导入
python -c "import torch, pybullet, pytorch_lightning; print('环境 OK')"


# =========================
# 1. 生成 Panda 动态障碍轨迹库
# =========================

# 这个文件会被真实 PyBullet 数据采集、ArmLidar 数据集和动态障碍评估使用
python -m neural_cbf.tools.generate_panda_obstacle_trajs \
  --num_traj 200 \
  --num_steps 300 \
  --dt 0.0166666667 \
  --out data/obstacle_trajs/panda_trajs.npz


# =========================
# 2. 生成新版 all-rays-at-once neural raycast surrogate 数据集
# =========================

# 默认双机械臂布局：面对面，ego=(0,-0.25,0)，obstacle=(0,0.25,0)，基座距离 0.5m
# 默认分层：碰撞/unsafe <0.01m，near 0.01-0.05m，mid 0.05-0.20m，far >0.20m
# 默认占比：collision:near:mid:far = 10:40:35:15

# 2.1 debug 采样：默认 10k states，6 anchors x 128 rays = 768 rays
python scripts/generate_neural_raycast_dataset.py \
  --preset debug \
  --out_dir dataset/neural_raycast_debug \
  --seed 0

# 2.2 full 采样：默认 100k states
python scripts/generate_neural_raycast_dataset.py \
  --preset full \
  --out_dir dataset/neural_raycast_full \
  --seed 0

# 2.3 自定义小样本快速冒烟测试
python scripts/generate_neural_raycast_dataset.py \
  --num_samples 200 \
  --out_dir dataset/neural_raycast_smoke \
  --seed 0


# =========================
# 3. 训练最新 FK-aware ray-link g_phi
# =========================

# 3.1 先检查 Torch FK 和采样阶段 PyBullet FK 是否一致
# 这个脚本只做一致性检查，不训练模型
python -m loss.tests.check_fk_consistency \
  --dataset_dir dataset/neural_raycast_full \
  --config loss/configs/config_raylink_g_phi.yaml \
  --num_samples 32 \
  --device cpu

# 3.2 训练 FK-aware neural raycast surrogate
# 输入是 q_ego, q_obs；模型内部先用 Torch FK 生成 ray-link geometry query
# 默认输出目录：loss/outputs_raylink_g_phi
python -m loss.training.train_raylink_g_phi \
  --config loss/configs/config_raylink_g_phi.yaml \
  --dataset_dir dataset/neural_raycast_full \
  --device cuda

# 3.3 用 best_safety checkpoint 在 test split 上评估
# 评估会输出 overall、near_boundary、collision_unsafe 等分组指标
python -m loss.eval.eval_raylink_g_phi \
  --config loss/configs/config_raylink_g_phi.yaml \
  --ckpt loss/outputs_raylink_g_phi/checkpoints/best_safety.pt \
  --dataset_dir dataset/neural_raycast_full \
  --split test \
  --out_dir loss/outputs_raylink_g_phi/eval \
  --device cuda


# =========================
# 4. neural_cbf/ Lightning baseline 控制器训练流程
# =========================

# FK-aware g_phi 现在只保留 ray-link 训练/eval 管线；旧 gphi-chain 控制器命令已删除。
# 把 RayLinkMLPGPhi 接到 h_theta/QP 前，不要再使用旧版 g_phi 命令。

# 4.1 单独生成 ArmLidar 数据集，不训练
# 数据会写到 datamodule 默认缓存路径；dataset_name 会参与文件名。
python scripts/generate_dataset_arm_lidar.py \
  --robot_name panda \
  --obstacle_robot_name panda \
  --obstacle_traj_path data/obstacle_trajs/panda_trajs.npz \
  --dataset_name pino_motor_norm \
  --observation_type uniform_surface \
  --point_dim 3 \
  --n_observation 256 \
  --n_observation_dataset 256 \
  --max_episode 120 \
  --trajectories_per_episode 40 \
  --trajectory_length 35 \
  --fixed_samples 200

# 4.2 训练 baseline：raw observation + finite-difference hdot
# 输出一般在 models/neural_cbf/<dynamics_name>/<version>/ 下。
python neural_cbf/training/train_arm_lidar.py \
  --robot_name panda \
  --obstacle_robot_name panda \
  --obstacle_traj_path data/obstacle_trajs/panda_trajs.npz \
  --version baseline_fd_raw \
  --baseline \
  --dataset_name pino_motor_norm \
  --observation_type uniform_surface \
  --point_dim 3 \
  --n_observation 256 \
  --n_observation_dataset 256 \
  --max_epochs 121 \
  --batch_size 64 \
  --devices 0


# =========================
# 5. 单模型离线评估和 rollout
# =========================

# 5.1 找训练出来的 Lightning checkpoint
find models -name "*.ckpt" | sort

# 5.2 单模型离线指标评估：baseline checkpoint
python neural_cbf/evaluation/eval_arm_lidar.py \
  --ckpt PATH/TO/baseline.ckpt \
  --mode metrics \
  --baseline \
  --obs_backend raw \
  --train_use_fd \
  --num_samples 2048 \
  --batch_size 256 \
  --seed 0 \
  --out results/baseline_metrics.json

# 5.3 图形界面 rollout，可视化动态障碍场景
python neural_cbf/evaluation/eval_arm_lidar.py \
  --ckpt PATH/TO/baseline.ckpt \
  --mode rollout \
  --gui \
  --baseline \
  --obs_backend raw \
  --train_use_fd \
  --t_sim 6.0 \
  --speed_scale 1.8 \
  --obstacle_mode arm_task \
  --scene cross_pick
