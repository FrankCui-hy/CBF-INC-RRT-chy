# =========================
# 2. 生成新版 all-rays-at-once neural raycast surrogate 数据集
# =========================

# 默认双机械臂布局：面对面，ego=(0,-0.25,0)，obstacle=(0,0.25,0)，基座距离 0.5m
# 默认分层：碰撞/unsafe <0.01m，near 0.01-0.05m，mid 0.05-0.20m，far >0.20m
# 默认占比：collision:near:mid:far = 10:40:35:15

# 2.1 debug 采样：默认 10k states，6 anchors x 128 rays = 768 rays
python scripts/generate_neural_raycast_dataset.py \
  --preset debug \
  --out_dir dataset/neural_raycast_debug \
  --seed 0

# 2.2 full 采样：默认 100k states
python scripts/generate_neural_raycast_dataset.py \
  --preset full \
  --out_dir dataset/neural_raycast_full \
  --seed 0

# 2.3 自定义小样本快速冒烟测试
python scripts/generate_neural_raycast_dataset.py \
  --num_samples 200 \
  --out_dir dataset/neural_raycast_smoke \
  --seed 0


# =========================
# 3. 训练最新 FK-aware ray-link g_phi
# =========================

# 3.1 先检查 Torch FK 和采样阶段 PyBullet FK 是否一致
# 这个脚本只做一致性检查，不训练模型
python -m loss.tests.check_fk_consistency \
  --dataset_dir dataset/neural_raycast_full \
  --config loss/configs/config_raylink_g_phi.yaml \
  --num_samples 32 \
  --device cpu

# 3.2 训练 FK-aware neural raycast surrogate
# 输入是 q_ego, q_obs；模型内部先用 Torch FK 生成 ray-link geometry query
# 默认输出目录：loss/outputs_raylink_g_phi
python -m loss.training.train_raylink_g_phi \
  --config loss/configs/config_raylink_g_phi.yaml \
  --dataset_dir dataset/neural_raycast_full \
  --device cuda

# 3.3 用 best_safety checkpoint 在 test split 上评估
# 评估会输出 overall、near_boundary、collision_unsafe 等分组指标
python -m loss.eval.eval_raylink_g_phi \
  --config loss/configs/config_raylink_g_phi.yaml \
  --ckpt loss/outputs_raylink_g_phi/checkpoints/best_safety.pt \
  --dataset_dir dataset/neural_raycast_full \
  --split test \
  --out_dir loss/outputs_raylink_g_phi/eval \
  --device cuda


# =========================
# 4. neural_cbf/ Lightning baseline 控制器训练流程
# =========================

# FK-aware g_phi 现在只保留 ray-link 训练/eval 管线；旧 gphi-chain 控制器命令已删除。
# 把 RayLinkMLPGPhi 接到 h_theta/QP 前，不要再使用旧版 g_phi 命令。

# 4.0 从 RayLink g_phi 数据集转换 CBF state-label cache
# 这一步复用 RayLink 数据里的 q_ego/q_obs/d_min/collision/depth：
# - safe/unsafe/boundary label 来自 d_min/collision
# - qdot_obs 如果没有真实动态轨迹，先显式设为 0（静态障碍 baseline）
# - --include_raylink_oracle_points 会离线缓存 RayLink oracle point cloud，供 raylink_cached_oracle FD baseline 训练使用
# 默认 RayLink layout 是 6 anchors x 128 rays = 768 points，所以这里必须用 --n_observation_dataset 768
python scripts/convert_raylink_raycast_to_cbf_state_label_cache.py \
  --dataset_dir dataset/neural_raycast_full \
  --out dataset/cbf_state_label/raylink_cached_oracle_state_label.pt \
  --gphi_ckpt loss/outputs_raylink_g_phi/checkpoints/best_safety.pt \
  --d_safe 0.02 \
  --qdot_mode zeros \
  --assume_static_obstacle \
  --include_raylink_oracle_points \
  --n_observation 256 \
  --n_observation_dataset 768 \
  --jacobian_chunk_size 512 \
  --raylink_points_chunk_size 512

# 4.0a Smoke test 四条 CBF observation 路径
# 这里会覆盖 legacy_oracle、gphi static、gphi dynamic、在线 raylink_oracle、cached raylink oracle。

  --gphi_ckpt loss/outputs_raylink_g_phi/checkpoints/best_safety.pt \

  --n_observation 32 \
  --n_observation_dataset 64

# 4.0b 正式训练 RayLink cached oracle FD baseline
# 这是和论文 baseline 对齐的路径：训练时不在每个 step 重新 PyBullet raycast。
# datax = q_ego + cached RayLink oracle point cloud + aux，FD 只通过 q perturb + local transform 估 hdot。
python neural_cbf/training/train_arm_lidar.py \
  --robot_name panda \
  --obstacle_robot_name panda \
  --obstacle_traj_path data/obstacle_trajs/panda_trajs.npz \
  --version cbf_raylink_cached_oracle_fd \

  --cbf_obs_mode raylink_cached_oracle \
  --gphi_ckpt loss/outputs_raylink_g_phi/checkpoints/best_safety.pt \
  --state_label_cache dataset/cbf_state_label/raylink_cached_oracle_state_label.pt \
  --train_use_fd \
  --dataset_name pino_motor \
  --observation_type uniform_surface \
  --point_dim 3 \
  --n_observation 256 \
  --n_observation_dataset 768 \
  --max_epochs 121 \
  --batch_size 64 \
  --devices 0

# 4.0c 训练 ours: frozen g_phi + analytic chain rule
# 使用同一个 state-label cache；gphi 模式会忽略 cached point cloud，在线调用 frozen g_phi 保持可导。

  --gphi_ckpt loss/outputs_raylink_g_phi/checkpoints/best_safety.pt \

  --no_train_use_fd \
  --dataset_name pino_motor \

  --n_observation_dataset 768 \
